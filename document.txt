This is IMPORTANT for demo & portfolio

🧠 Logic Rules

Current bid

Bid history

No backend, only React state

🧪 Dummy Product Data
const productData = {
  id: 1,
  name: "Sp Product – Premium Almonds",
  category: "Dry Fruits",
  basePrice: 500,
  currentBid: 850,
  bids: [
    { user: "User123", amount: 850 },
    { user: "User421", amount: 800 },
    { user: "User009", amount: 750 }
  ]
};

⚙️ Bid State Setup (ProductDetails.jsx)
const [currentBid, setCurrentBid] = useState(productData.currentBid);
const [bidHistory, setBidHistory] = useState(productData.bids);
const [bidAmount, setBidAmount] = useState("");

✅ Place Bid Handler (CORE LOGIC)
const placeBid = () => {
  if (Number(bidAmount) <= currentBid) {
    alert("Bid must be higher than current bid");
    return;
  }

  setCurrentBid(Number(bidAmount));

  setBidHistory([
    { user: "You", amount: Number(bidAmount) },
    ...bidHistory
  ]);

  setBidAmount("");
};

⏱ Countdown Timer (Dummy Auction)
const [timeLeft, setTimeLeft] = useState(2 * 60 * 60); // 2 hours

useEffect(() => {
  const timer = setInterval(() => {
    setTimeLeft(prev => prev > 0 ? prev - 1 : 0);
  }, 1000);

  return () => clearInterval(timer);
}, []);

HH:MM:SS
